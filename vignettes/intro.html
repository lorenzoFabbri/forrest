Introduction to forrest

forrest() draws a forest plot from any data frame with one row per estimate and columns for the point estimate and the confidence limits. The rows can be regression coefficients, subgroup estimates, studies in a meta-analysis or exposure categories. The package’s only dependency is tinyplot, and it accepts data frames, tibbles and data.tables.

This vignette shows each feature on a small example. The data in every example are invented for illustration.

Basic forest plot

A call needs three column names: estimate, lower and upper. Here they hold coefficients from a linear model of systolic blood pressure (SBP).

dat <- data.frame(
  predictor = c("Age (per 10 y)", "Female sex", "BMI (per 5 kg/m²)",
                "Current smoker", "Physically active"),
  estimate  = c( 0.18, -0.42,  0.11, -0.31,  0.24),
  lower     = c( 0.05, -0.61, -0.04, -0.52,  0.08),
  upper     = c( 0.31, -0.23,  0.26, -0.10,  0.40)
)

forrest(
  dat,
  estimate = "estimate",
  lower    = "lower",
  upper    = "upper",
  label    = "predictor",
  xlab     = "Regression coefficient (95% CI)"
)

Section headers from a grouping column

When section names a column, forrest() inserts a bold header row wherever that column’s value changes, indents the labels beneath it, and adds a blank row after each section. Rows stay in the order they have in the data, so sort the data first if the sections should appear in a particular order.

sub_dat <- data.frame(
  subgroup = c("Sex",    "Sex",
               "Age group", "Age group", "Age group"),
  label    = c("Female", "Male",
               "30–49 years", "50–69 years", "70+ years"),
  estimate = c(-0.38,  0.12,  0.22, -0.15, -0.41),
  lower    = c(-0.58, -0.08,  0.02, -0.38, -0.66),
  upper    = c(-0.18,  0.32,  0.42,  0.08, -0.16)
)

forrest(
  sub_dat,
  estimate = "estimate",
  lower    = "lower",
  upper    = "upper",
  label    = "label",
  section  = "subgroup",
  xlab     = "Regression coefficient (95% CI)",
  header   = "Subgroup"
)

section_indent = FALSE turns off the indentation and section_spacer = FALSE removes the blank row after each section.

    "Green space (%)", "Walkability", "Noise (per 10 dB)",

    "Area deprivation", "Employment rate"

Summary estimates

Rows where the is_summary column is TRUE are drawn as filled diamonds spanning the confidence interval, the usual mark for a pooled or overall estimate.

sex_dat <- data.frame(
  label    = c("Female", "Male", "Overall"),
  estimate = c(-0.42, -0.29, -0.36),
  lower    = c(-0.61, -0.48, -0.50),
  upper    = c(-0.23, -0.10, -0.22),
  is_sum   = c(FALSE, FALSE, TRUE)
)

forrest(
  sex_dat,
  estimate   = "estimate",
  lower      = "lower",
  upper      = "upper",
  label      = "label",
  is_summary = "is_sum",
  xlab       = "Regression coefficient (95% CI)",
  title      = "Association of physical activity with SBP, by sex"
)

Group colours

A group column colours each row by its value and adds a legend. The default palette is Okabe-Ito, which stays distinguishable under the common forms of colour blindness. To choose the colours yourself, pass col, either as an unnamed vector used in order of appearance or as a vector named by group level.

grp_dat <- data.frame(
  predictor = c("Air pollution (PM2.5)", "Noise exposure",
                "Green space access", "Walkability index",
                "Food environment",
                "Social cohesion", "Social isolation",
                "Area deprivation", "Neighbourhood safety",
                "Employment rate"),
  domain    = rep(c("Physical environment", "Social environment"),
                  each = 5),
  estimate  = c(-0.18,  0.12, -0.22,  0.15, -0.08,
                 0.05, -0.03,  0.09, -0.11,  0.14),
  lower     = c(-0.38, -0.08, -0.42, -0.05, -0.28,
                -0.12, -0.20, -0.09, -0.31, -0.04),
  upper     = c( 0.02,  0.32, -0.02,  0.35,  0.12,
                 0.22,  0.14,  0.27,  0.09,  0.32)
)

forrest(
  grp_dat,
  estimate = "estimate",
  lower    = "lower",
  upper    = "upper",
  label    = "predictor",
  group    = "domain",
  xlab     = "Mean difference in SBP (mmHg, 95% CI)"
)

Several estimates per row

With dodge = TRUE, consecutive rows that share a label value are drawn as one band: their intervals are offset vertically around the band’s centre and the label appears once. Pair it with group so the reader can tell the series apart.

dodge_dat <- data.frame(
  exposure = rep(
    c("PM2.5 (per 10 μg/m³)", "NO2 (per 10 ppb)",
      "Noise (per 10 dB)", "Green space (%)", "Walkability"),
    each = 2
  ),
  period   = rep(c("Childhood", "Adulthood"), 5),
  estimate = c(
     0.14, -0.05,  0.08,  0.12, -0.19, -0.06,  0.11, -0.03,  0.07,  0.10
  ),
  lower    = c(
    -0.10, -0.26, -0.09, -0.08, -0.40, -0.25, -0.05, -0.12, -0.14, -0.09
  ),
  upper    = c(
     0.38,  0.16,  0.25,  0.32,  0.02,  0.13,  0.27,  0.06,  0.28,  0.29
  )
)

forrest(
  dodge_dat,
  estimate = "estimate",
  lower    = "lower",
  upper    = "upper",
  label    = "exposure",
  group    = "period",
  dodge    = TRUE,
  header   = "Environmental exposure",
  xlab     = "Mean difference in SBP (mmHg, 95% CI)"
)

A number instead of TRUE sets the distance between neighbouring intervals in a band, in units of one row; TRUE means 0.25. Section and subsection headers and spacer rows are never merged into a band.

One text column per series

By default each text column in cols prints a value beside every interval, at that interval’s offset position. With cols_by_group = TRUE, each text column instead prints one value per band, at the band’s centre: the first non-empty value among the band’s rows. Giving each series its own column, filled only on that series’ rows, produces a wide table with one row per label and one column per series.

dodge_dat$est_ci     <- sprintf("%.2f (%.2f, %.2f)",
                                dodge_dat$estimate,
                                dodge_dat$lower,
                                dodge_dat$upper)
dodge_dat$text_child <- ifelse(
  dodge_dat$period == "Childhood", dodge_dat$est_ci, ""
)
dodge_dat$text_adult <- ifelse(
  dodge_dat$period == "Adulthood", dodge_dat$est_ci, ""
)

forrest(
  dodge_dat,
  estimate      = "estimate",
  lower         = "lower",
  upper         = "upper",
  label         = "exposure",
  group         = "period",
  dodge         = TRUE,
  cols_by_group = TRUE,
  cols          = c("Childhood (95% CI)" = "text_child",
                    "Adulthood (95% CI)" = "text_adult"),
  widths        = c(2.8, 3.5, 2.2, 2.2),
  header        = "Environmental exposure",
  xlab          = "Mean difference in SBP (mmHg, 95% CI)"
)

Point shapes

A shape column gives each of its values a different point character and adds a second legend. Combined with group and dodge, a single band can then encode two variables, here period by colour and sex by shape.

shape_dat <- data.frame(
  exposure = rep(
    c("PM2.5 (per 10 μg/m³)", "NO2 (per 10 ppb)",
      "Noise (per 10 dB)", "Green space (%)", "Walkability"),
    each = 4
  ),
  period   = rep(rep(c("Childhood", "Adulthood"), each = 2), 5),
  sex      = rep(c("Female", "Male"), 10),
  estimate = c(
     0.22, -0.08, -0.10,  0.18,
     0.11,  0.05,  0.00,  0.14,
    -0.31,  0.06, -0.09, -0.02,
     0.08,  0.12, -0.04,  0.03,
     0.17, -0.06,  0.22,  0.01
  ),
  lower    = c(
    -0.20, -0.38, -0.28, -0.14,
    -0.12, -0.22, -0.22, -0.14,
    -0.56, -0.24, -0.38, -0.30,
    -0.18, -0.14, -0.22, -0.15,
    -0.09, -0.38, -0.04, -0.27
  ),
  upper    = c(
     0.64,  0.22,  0.08,  0.50,
     0.34,  0.32,  0.22,  0.42,
    -0.06,  0.36,  0.20,  0.26,
     0.34,  0.38,  0.14,  0.21,
     0.43,  0.26,  0.48,  0.29
  )
)

forrest(
  shape_dat,
  estimate = "estimate",
  lower    = "lower",
  upper    = "upper",
  label    = "exposure",
  group    = "period",
  shape    = "sex",
  dodge    = TRUE,
  xlab     = "Mean difference in SBP (mmHg, 95% CI)"
)

The shape legend sits at legend_shape_pos, "bottomright" by default, and legend_shape_pos = NULL removes it. legend_pos does the same for the colour legend.

Text columns

cols adds text columns to the right of the plot. It is a named character vector: each name is a column header and each value is the name of a column in data. Format the text yourself, for instance with sprintf(), so that it shows exactly the digits you want. widths sets the relative widths of the label panel, the plot and each text column.

tc_dat <- data.frame(
  predictor = c("Age (per 10 y)", "Female sex", "BMI (per 5 kg/m²)",
                "Current smoker", "Physically active"),
  estimate  = c( 0.18, -0.42,  0.11, -0.31,  0.24),
  lower     = c( 0.05, -0.61, -0.04, -0.52,  0.08),
  upper     = c( 0.31, -0.23,  0.26, -0.10,  0.40),
  coef_ci   = c(
    " 0.18 ( 0.05,  0.31)",
    "-0.42 (-0.61, -0.23)",
    " 0.11 (-0.04,  0.26)",
    "-0.31 (-0.52, -0.10)",
    " 0.24 ( 0.08,  0.40)"
  ),
  pval = c("0.006", "<0.001", "0.148", "0.003", "0.009")
)

forrest(
  tc_dat,
  estimate = "estimate",
  lower    = "lower",
  upper    = "upper",
  label    = "predictor",
  header   = "Predictor",
  cols     = c("Coef (95% CI)" = "coef_ci", "P-value" = "pval"),
  widths   = c(2.8, 4, 2.5, 1.2),
  xlab     = "Regression coefficient (95% CI)"
)

Values in section header rows

Text columns are empty on the header rows that section inserts. section_cols fills chosen text columns on those rows with a section-level value. Its names must be names of cols, and its values are columns in data; on each header row, the named text column shows the first non-empty value of that data column within the section. subsection_cols does the same for subsection headers. Here each row shows its own sample size and each header shows the section’s total.

sc_dat <- data.frame(
  subgroup  = c("Sex",    "Sex",   "Age group", "Age group", "Age group"),
  label     = c("Female", "Male",  "30–49 y", "50–69 y", "70+ y"),
  estimate  = c(-0.38,  0.12,  0.22, -0.15, -0.41),
  lower     = c(-0.58, -0.08,  0.02, -0.38, -0.66),
  upper     = c(-0.18,  0.32,  0.42,  0.08, -0.16),
  coef_ci   = c(
    "-0.38 (-0.58, -0.18)",
    " 0.12 (-0.08,  0.32)",
    " 0.22 ( 0.02,  0.42)",
    "-0.15 (-0.38,  0.08)",
    "-0.41 (-0.66, -0.16)"
  ),
  n_text    = c("1,204", "1,096", "702", "958", "640"),
  n_section = c("2,300", "", "2,300", "", "")
)

forrest(
  sc_dat,
  estimate     = "estimate",
  lower        = "lower",
  upper        = "upper",
  label        = "label",
  section      = "subgroup",
  section_cols = c("N" = "n_section"),
  header       = "Subgroup",
  cols         = c("Coef (95% CI)" = "coef_ci", "N" = "n_text"),
  widths       = c(2.5, 4, 2.5, 1.2),
  xlab         = "Regression coefficient (95% CI)"
)

Row stripes

stripe = TRUE shades every other row, which helps the eye follow a row across a wide plot.

stripe_dat <- data.frame(
  label    = c(
    "Age (per 10 y)", "Female sex", "BMI (per 5 kg/m²)",
    "Current smoker", "Physically active",
    "Alcohol intake", "Sleep duration", "Depressive symptoms"
  ),
  estimate = c( 0.42, -0.18,  0.31, -0.07,  0.25, -0.12,  0.19, -0.34),
  lower    = c( 0.22, -0.38,  0.12, -0.24,  0.06, -0.30,  0.01, -0.52),
  upper    = c( 0.62,  0.02,  0.50,  0.10,  0.44,  0.06,  0.37, -0.16)
)

forrest(
  stripe_dat,
  estimate = "estimate",
  lower    = "lower",
  upper    = "upper",
  label    = "label",
  stripe   = TRUE,
  xlab     = "Regression coefficient (95% CI)"
)

Themes

theme takes the name of a built-in theme ("default", "minimal" or "classic") or a named list that overrides individual keys of the default theme. The keys are grid_col, grid_lty and grid_lwd for the horizontal gridlines, ref_col and ref_lty for the reference line, and stripe_col for striped rows.

theme_dat <- data.frame(
  predictor = c("Age (per 10 y)", "Female sex", "BMI (per 5 kg/m²)"),
  estimate  = c( 0.18, -0.42,  0.11),
  lower     = c( 0.05, -0.61, -0.04),
  upper     = c( 0.31, -0.23,  0.26)
)

# Lighter gridlines and a softer reference line
forrest(
  theme_dat,
  estimate = "estimate",
  lower    = "lower",
  upper    = "upper",
  label    = "predictor",
  theme    = "minimal",
  title    = 'theme = "minimal"',
  xlab     = "Coefficient (95% CI)"
)
# Dotted gridlines and a solid black reference line
forrest(
  theme_dat,
  estimate = "estimate",
  lower    = "lower",
  upper    = "upper",
  label    = "predictor",
  theme    = "classic",
  title    = 'theme = "classic"',
  xlab     = "Coefficient (95% CI)"
)
forrest(
  theme_dat,
  estimate = "estimate",
  lower    = "lower",
  upper    = "upper",
  label    = "predictor",
  theme    = list(ref_col = "#e63946", ref_lty = 1L, grid_col = "#eeeeee"),
  title    = "Custom theme (red reference line)",
  xlab     = "Coefficient (95% CI)"
)

Graphical parameters that forrest() does not name, such as cex.axis and cex.lab, are passed through ... to the call that sets up the plot region.

Saving plots

save_forrest() writes a plot to PDF, PNG, SVG or TIFF, choosing the device from the file extension. It takes a function with no arguments that draws the plot, so the device is opened before drawing starts and closed even if drawing fails. dpi sets the resolution of PNG and TIFF files.

save_forrest(
  file   = "my_forest_plot.pdf",
  plot   = function() {
    forrest(
      dat,
      estimate = "estimate",
      lower    = "lower",
      upper    = "upper",
      label    = "predictor",
      xlab     = "Regression coefficient (95% CI)"
    )
  },
  width  = 8,
  height = 5
)
